    // --- 3. MC ロジック ---

    function initMC() {

      // 部屋の初期状態同期

      db.collection("game").doc("state").onSnapshot(doc => {

        if (!doc.exists) return;

        const state = doc.data();

        const qIndex = state.currentQIndex || 0;

        const q = quizQuestions[qIndex];


        document.getElementById("mc-q-num").textContent = `第 ${qIndex + 1} 問 / 全 ${quizQuestions.length} 問`;

        document.getElementById("mc-q-answers").textContent = `想定解: ${q.answers.join(", ")}`;

        document.getElementById("mc-status-badge").textContent = 

          state.status === "open" ? "回答受付中" : state.status === "closed" ? "締切" : "待機中";

      });


      // 回答・状況のリアルタイム監視

      db.collection("answers").onSnapshot(snapshot => {

        const answers = {};

        snapshot.forEach(doc => { answers[doc.id] = doc.data(); });

        renderMCTable(answers);

      });


      // スコアのリアルタイム監視

      db.collection("scores").onSnapshot(snapshot => {

        const scores = {};

        snapshot.forEach(doc => { scores[doc.id] = doc.data().score || 0; });

        renderMCLeaderboard(scores);

      });

    }


    function renderMCTable(answers) {

      const tbody = document.getElementById("mc-answers-table");

      tbody.innerHTML = "";


      TEAMS.forEach(team => {

        const data = answers[team] || { text: "", isPublic: false, isDuplicate: false, isCorrect: false };

        const tr = document.createElement("tr");

        tr.className = "border-b text-sm";

        tr.innerHTML = `

          <td class="p-2 font-bold">${team}班</td>

          <td class="p-2">${data.text || '<span class="text-gray-400">未送信</span>'}</td>

          <td class="p-2 text-center">

            <input type="checkbox" ${data.isPublic ? 'checked' : ''} onchange="updateAnswerFlag('${team}', 'isPublic', this.checked)">

          </td>

          <td class="p-2 text-center">

            <input type="checkbox" ${data.isDuplicate ? 'checked' : ''} onchange="updateAnswerFlag('${team}', 'isDuplicate', this.checked)">

          </td>

          <td class="p-2 text-center">

            <input type="checkbox" ${data.isCorrect ? 'checked' : ''} onchange="updateAnswerFlag('${team}', 'isCorrect', this.checked)">

          </td>

        `;

      });

    async function updateAnswerFlag(team, field, value) {

      await db.collection("answers").doc(team).set({ [field]: value }, { merge: true });

    // カブり・自動正誤判定ロジック

    async function checkDuplicates() {

      const stateDoc = await db.collection("game").doc("state").get();

      const qIndex = stateDoc.data().currentQIndex || 0;

      const validAnswers = quizQuestions[qIndex].answers.map(a => a.trim().toLowerCase());


      const snapshot = await db.collection("answers").get();

      const countMap = {};

      const teamAnswers = {};


      snapshot.forEach(doc => {

        const text = (doc.data().text || "").trim().toLowerCase();

        teamAnswers[doc.id] = text;

        if (text) countMap[text] = (countMap[text] || 0) + 1;

      });


      TEAMS.forEach(async team => {

        const ans = teamAnswers[team];

        const isDuplicate = ans ? countMap[ans] > 1 : false;

        const isCorrect = ans ? validAnswers.includes(ans) : false;


        await db.collection("answers").doc(team).set({

          isDuplicate: isDuplicate,

          isCorrect: isCorrect && !isDuplicate // カブっていない正解のみ自動正解扱い

        }, { merge: true });

      });

    }


    async function finalizeScore() {

      const snapshot = await db.collection("answers").get();

      snapshot.forEach(async doc => {

        const data = doc.data();

        if (data.isCorrect && !data.isDuplicate) {

          const scoreDoc = await db.collection("scores").doc(doc.id).get();

          const currentScore = (scoreDoc.exists ? scoreDoc.data().score : 0) || 0;

          await db.collection("scores").doc(doc.id).set({ score: currentScore + 1 });

        }

      });

      alert("ポイントの加算が完了しました。");

    }


    async function nextQuestion() {

      const stateDoc = await db.collection("game").doc("state").get();

      const nextIndex = (stateDoc.data().currentQIndex || 0) + 1;

      

      if (nextIndex >= quizQuestions.length) {

        alert("全問題が終了しました。");

        return;

      }


      // 次の問題へ進み、各班の回答フラグをリセット

      await db.collection("game").doc("state").set({ currentQIndex: nextIndex, status: "waiting" });

      TEAMS.forEach(async team => {

        await db.collection("answers").doc(team).set({ text: "", isPublic: false, isDuplicate: false, isCorrect: false });

      });

    }

    function initPlayer() {

      // 部屋の状況をリアルタイム同期

      db.collection("game").doc("state").onSnapshot(doc => {

        if (!doc.exists) return;

        const state = doc.data();

        const q = quizQuestions[qIndex];


        document.getElementById("player-q-num").textContent = `第 ${qIndex + 1} 問`;

        document.getElementById("player-q-text").textContent = q.text;


        const isClosed = state.status !== "open";

        document.getElementById("player-input").disabled = isClosed;

        document.getElementById("player-submit-btn").disabled = isClosed;

        document.getElementById("player-status-badge").textContent = 

          state.status === "open" ? "受付中" : "受付停止中";

      });


      // 公開された回答をリアルタイム同期

      db.collection("answers").onSnapshot(snapshot => {

        const answers = {};

        snapshot.forEach(doc => { answers[doc.id] = doc.data(); });


        // 自分の送信状態確認

        if (answers[myTeam] && answers[myTeam].text) {

          document.getElementById("player-input").value = answers[myTeam].text;

          document.getElementById("player-input").disabled = true;

          document.getElementById("player-submit-btn").disabled = true;

          document.getElementById("player-submitted-msg").classList.remove("hidden");

        } else {

          document.getElementById("player-submitted-msg").classList.add("hidden");

        }


        renderPlayerPublicAnswers(answers);

      });

    }


    function renderPlayerPublicAnswers(answers) {

      const container = document.getElementById("player-public-answers");

      container.innerHTML = "";


      TEAMS.forEach(team => {

        const data = answers[team];

        if (data && data.isPublic) {

          const div = document.createElement("div");

          const isDup = data.isDuplicate;

          div.className = `p-2 rounded border font-bold text-center ${isDup ? 'bg-rose-100 text-rose-800 border-rose-300' : 'bg-slate-50 border-slate-200'}`;

          div.textContent = `${team}班: ${data.text} ${isDup ? '(カブり!)' : ''}`;

          container.appendChild(div);

        }

      });

    }


    async function submitAnswer() {

      const text = document.getElementById("player-input").value.trim();

      if (!text) return;


      await db.collection("answers").doc(myTeam).set({

        text: text,

        isPublic: false,

        isDuplicate: false,

        isCorrect: false

      }, { merge: true });

    }
